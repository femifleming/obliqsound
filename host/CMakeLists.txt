cmake_minimum_required(VERSION 3.22)
project(KiaHost VERSION 1.0)
set(CMAKE_OSX_ARCHITECTURES arm64)
add_subdirectory($ENV{HOME}/cassetter-plugin/JUCE ${CMAKE_BINARY_DIR}/juce)
juce_add_gui_app(KiaHost PRODUCT_NAME "KiaHost")
target_sources(KiaHost PRIVATE Host.mm)
target_compile_definitions(KiaHost PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(KiaHost PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
target_link_libraries(KiaHost PRIVATE "-framework AppKit")
