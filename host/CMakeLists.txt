cmake_minimum_required(VERSION 3.22)
project(ObliqHost VERSION 1.0)
set(CMAKE_OSX_ARCHITECTURES arm64)
add_subdirectory($ENV{HOME}/cassetter-plugin/JUCE ${CMAKE_BINARY_DIR}/juce)
juce_add_gui_app(ObliqHost PRODUCT_NAME "ObliqHost")
target_sources(ObliqHost PRIVATE Host.mm)
target_compile_definitions(ObliqHost PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(ObliqHost PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
target_link_libraries(ObliqHost PRIVATE "-framework AppKit")
